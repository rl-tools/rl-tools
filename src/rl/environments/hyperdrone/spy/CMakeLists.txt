set(RL_TOOLS_ENABLE_HYPERDRONE_SPY OFF PARENT_SCOPE)
if(RL_TOOLS_DISABLE_HYPERDRONE_SPY)
    return()
endif()
if(NOT RL_TOOLS_RENDERING_ENABLE_RAYTRACING)
    set(RL_TOOLS_SUMMARY_HYPERDRONE_SPY_REASON "raytracing unavailable" PARENT_SCOPE)
    return()
endif()

find_package(glfw3 QUIET)
find_package(OpenGL QUIET)
if(NOT TARGET glfw OR NOT TARGET OpenGL::GL)
    set(RL_TOOLS_SUMMARY_HYPERDRONE_SPY_REASON "GLFW/OpenGL not found (e.g. apt install libglfw3-dev libgl1-mesa-dev)" PARENT_SCOPE)
    return()
endif()

set(RL_TOOLS_IMGUI_VERSION 1.92.9b)
find_package(Git QUIET)
if(NOT ((GIT_FOUND AND NOT RL_TOOLS_OFFLINE_BUILD) OR EXISTS "${FETCHCONTENT_BASE_DIR}/imgui-src" OR FETCHCONTENT_SOURCE_DIR_IMGUI))
    set(RL_TOOLS_SUMMARY_HYPERDRONE_SPY_REASON "cannot fetch Dear ImGui (offline/no git)" PARENT_SCOPE)
    return()
endif()
FetchContent_Declare(imgui
    GIT_REPOSITORY https://github.com/ocornut/imgui.git
    GIT_TAG        v${RL_TOOLS_IMGUI_VERSION}
    GIT_SHALLOW    TRUE
)
rl_tools_fetchcontent_makeavailable_quiet(imgui)

find_package(Threads REQUIRED)
add_library(rl_tools_hyperdrone_spy SHARED ui.cpp "${imgui_SOURCE_DIR}/imgui.cpp" "${imgui_SOURCE_DIR}/imgui_draw.cpp" "${imgui_SOURCE_DIR}/imgui_tables.cpp" "${imgui_SOURCE_DIR}/imgui_widgets.cpp" "${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp" "${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp")
add_library(RLtools::HyperDroneSpy ALIAS rl_tools_hyperdrone_spy)
target_link_libraries(rl_tools_hyperdrone_spy PUBLIC RLtools::Core Threads::Threads PRIVATE glfw OpenGL::GL ${CMAKE_DL_LIBS})
target_include_directories(rl_tools_hyperdrone_spy SYSTEM PRIVATE "${imgui_SOURCE_DIR}" "${imgui_SOURCE_DIR}/backends")
target_compile_definitions(rl_tools_hyperdrone_spy PUBLIC RL_TOOLS_ENABLE_HYPERDRONE_SPY=1)
set_target_properties(rl_tools_hyperdrone_spy PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(NOT RL_TOOLS_DISABLE_TARGETS)
    add_executable(example_hyperdrone_spy example.cpp)
    target_link_libraries(example_hyperdrone_spy PRIVATE RLtools::RLtools RLtools::HyperDroneSpy rendering_raytracing_backend rendering_raytracing_stb)
endif()
set(RL_TOOLS_ENABLE_HYPERDRONE_SPY ON PARENT_SCOPE)
set(RL_TOOLS_SUMMARY_HYPERDRONE_SPY_REASON "GLFW/OpenGL, Dear ImGui ${RL_TOOLS_IMGUI_VERSION} via FetchContent" PARENT_SCOPE)
