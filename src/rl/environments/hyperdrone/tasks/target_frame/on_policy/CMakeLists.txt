foreach(PROFILE production smoke)
    set(TARGET_NAME rl_environments_hyperdrone_tasks_target_frame_on_policy_cuda)
    if(PROFILE STREQUAL "smoke")
        string(APPEND TARGET_NAME _smoke)
    endif()
    add_executable(${TARGET_NAME} training.cu)
    target_compile_definitions(${TARGET_NAME} PRIVATE RL_TOOLS_HYPERDRONE_TARGET_NAME="${TARGET_NAME}")
    target_link_libraries(${TARGET_NAME} PRIVATE RLtools::RLtools rendering_raytracing_backend rendering_raytracing_stb)
    set_target_properties(${TARGET_NAME} PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    if(PROFILE STREQUAL "smoke")
        set_target_properties(${TARGET_NAME} PROPERTIES EXCLUDE_FROM_ALL TRUE)
        target_compile_definitions(${TARGET_NAME} PRIVATE RL_TOOLS_HYPERDRONE_ON_POLICY_SMOKE)
    endif()
endforeach()
