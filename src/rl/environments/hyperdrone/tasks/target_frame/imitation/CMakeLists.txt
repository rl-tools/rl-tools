if(NOT RL_TOOLS_HYPERDRONE_IMITATION_N_RANKS MATCHES "^(1|2)$")
    message(FATAL_ERROR "RL_TOOLS_HYPERDRONE_IMITATION_N_RANKS must be 1 or 2 to partition the two active Worlds")

    set(CUDA_COMPUTE ON)
endif()

function(rl_tools_add_hyperdrone_imitation SUFFIX USE_CUDA)
    set(TARGET_NAME rl_environments_hyperdrone_tasks_target_frame_${SUFFIX})
    if(USE_CUDA)
        set(SOURCE_FILE ${CMAKE_CURRENT_BINARY_DIR}/${SUFFIX}.cu)
        file(CONFIGURE OUTPUT ${SOURCE_FILE} CONTENT "#include \"${CMAKE_CURRENT_SOURCE_DIR}/training.cpp\"\n")
    else()
        set(SOURCE_FILE training.cpp)
    endif()
    add_executable(${TARGET_NAME} ${SOURCE_FILE})
    target_compile_definitions(${TARGET_NAME} PRIVATE RL_TOOLS_HYPERDRONE_TARGET_NAME="${TARGET_NAME}")
    target_link_libraries(${TARGET_NAME} PRIVATE RLtools::RLtools rendering_raytracing_backend rendering_raytracing_stb)
    if(USE_CUDA)
        set_target_properties(${TARGET_NAME} PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON CUDA_SEPARABLE_COMPILATION ON)
        target_compile_definitions(${TARGET_NAME} PRIVATE RL_TOOLS_HYPERDRONE_IMITATION_N_RANKS=${RL_TOOLS_HYPERDRONE_IMITATION_N_RANKS})
    else()
        target_compile_definitions(${TARGET_NAME} PRIVATE RL_TOOLS_HYPERDRONE_IMITATION_N_RANKS=1)
    endif()
    target_compile_definitions(${TARGET_NAME} PRIVATE ${ARGN})
endfunction()

rl_tools_add_hyperdrone_imitation(imitation ${CUDA_COMPUTE})
rl_tools_add_hyperdrone_imitation(imitation_small ${CUDA_COMPUTE} RL_TOOLS_HYPERDRONE_IMITATION_SMALL)
rl_tools_add_hyperdrone_imitation(state_estimation ${CUDA_COMPUTE} RL_TOOLS_HYPERDRONE_IMITATION_STATE_ESTIMATION)
rl_tools_add_hyperdrone_imitation(state_estimation_small ${CUDA_COMPUTE} RL_TOOLS_HYPERDRONE_IMITATION_STATE_ESTIMATION RL_TOOLS_HYPERDRONE_IMITATION_SMALL)

endif()
