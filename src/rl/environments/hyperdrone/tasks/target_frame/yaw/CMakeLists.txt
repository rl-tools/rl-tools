if(NOT RL_TOOLS_BACKEND_ENABLE_CUDNN OR NOT RL_TOOLS_ENABLE_HDF5 OR NOT RL_TOOLS_ENABLE_JSON)
    message(STATUS "Hyperdrone yaw training skipped: cuDNN, HDF5 and JSON are required")
    return()
endif()

set(YAW_TEACHER_HEADER "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../../nn_models/port_checkpoint/raptor/policy.h")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${YAW_TEACHER_HEADER}")
file(SHA256 "${YAW_TEACHER_HEADER}" YAW_TEACHER_SHA256)

foreach(PROFILE production smoke)
    set(TARGET_NAME rl_environments_hyperdrone_tasks_target_frame_yaw_cuda)
    if(PROFILE STREQUAL "smoke")
        string(APPEND TARGET_NAME _smoke)
    endif()
    add_executable(${TARGET_NAME} training.cu)
    target_compile_definitions(${TARGET_NAME} PRIVATE RL_TOOLS_HYPERDRONE_TARGET_NAME="${TARGET_NAME}")
    target_compile_definitions(${TARGET_NAME} PRIVATE RL_TOOLS_HYPERDRONE_YAW_TEACHER_SHA256="${YAW_TEACHER_SHA256}")
    target_link_libraries(${TARGET_NAME} PRIVATE RLtools::RLtools hyperlux_backend hyperlux_stb)
    if(TARGET RLtools::HyperLuxSpy)
        target_link_libraries(${TARGET_NAME} PRIVATE RLtools::HyperLuxSpy)
    endif()
    set_target_properties(${TARGET_NAME} PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    if(PROFILE STREQUAL "smoke")
        set_target_properties(${TARGET_NAME} PROPERTIES EXCLUDE_FROM_ALL TRUE)
        target_compile_definitions(${TARGET_NAME} PRIVATE RL_TOOLS_HYPERDRONE_YAW_SMOKE)
    endif()
endforeach()
