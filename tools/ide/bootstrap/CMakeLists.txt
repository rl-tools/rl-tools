cmake_minimum_required(VERSION 3.24)
project(rl_tools_ide_host_compiler NONE)
if(CMAKE_CONFIGURATION_TYPES)
    message(FATAL_ERROR "Use a single-configuration generator, such as Ninja or Unix Makefiles.")
endif()

get_filename_component(IDE_CMAKE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../toolchain/cmake" ABSOLUTE)
set(IDE_LOCK_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../dependencies.json")
include("${IDE_CMAKE_DIR}/sources.cmake")
include("${IDE_CMAKE_DIR}/external.cmake")
set(RL_TOOLS_IDE_JOBS 5 CACHE STRING "Parallel build jobs, at most 5")
if(NOT RL_TOOLS_IDE_JOBS MATCHES "^[1-5]$")
    message(FATAL_ERROR "RL_TOOLS_IDE_JOBS must be 1 through 5")
endif()
get_filename_component(RL_TOOLS_REPOSITORY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
file(READ "${IDE_LOCK_FILE}" lock)
string(JSON RL_TOOLS_IDE_TARGET GET "${lock}" target)
ide_fetch_sources(source_revisions "${IDE_LOCK_FILE}" llvm-project)
string(JSON revision GET "${source_revisions}" llvm-project commit)
string(JSON source GET "${source_revisions}" llvm-project directory)
if(EXISTS "${CMAKE_BINARY_DIR}/source-revision")
    file(READ "${CMAKE_BINARY_DIR}/source-revision" previous_revision)
    if(NOT previous_revision STREQUAL revision)
        file(REMOVE_RECURSE "${CMAKE_BINARY_DIR}/host")
    endif()
endif()
file(CONFIGURE OUTPUT "${CMAKE_BINARY_DIR}/source-revision" CONTENT "${revision}" @ONLY)
file(CONFIGURE OUTPUT "${CMAKE_BINARY_DIR}/source-revisions.json" CONTENT "${source_revisions}" @ONLY)
ide_check_sources("${CMAKE_BINARY_DIR}/source-revisions.json")
find_program(IDE_GIT git REQUIRED)
ide_run(IDE_SOURCE_DATE_EPOCH "${IDE_GIT}" -C "${source}" show -s --format=%ct HEAD)
ide_run(git_dir "${IDE_GIT}" -C "${source}" rev-parse --absolute-git-dir)
set(IDE_CONFIGURE_DEPENDS "${IDE_LOCK_FILE}" "${git_dir}/HEAD" "${IDE_CMAKE_DIR}/external.cmake" "${IDE_CMAKE_DIR}/llvm_options.cmake")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${IDE_CONFIGURE_DEPENDS})
include("${IDE_CMAKE_DIR}/llvm_options.cmake")
add_custom_target(check-source
    COMMAND "${CMAKE_COMMAND}" "-DREVISION_FILE=${CMAKE_BINARY_DIR}/source-revisions.json" -P "${IDE_CMAKE_DIR}/sources.cmake" VERBATIM)
ide_external(host-compiler SOURCE "${source}/llvm" BINARY "${CMAKE_BINARY_DIR}/host"
    ARGS ${IDE_LLVM_COMMON_ARGS} -DCMAKE_BUILD_TYPE=Release "-DLLVM_ENABLE_PROJECTS=clang|lld"
    TARGETS clang clang-resource-headers lld llvm-ar llvm-ranlib llvm-nm
    DEPENDS check-source
    BYPRODUCTS "${CMAKE_BINARY_DIR}/host/bin/clang" "${CMAKE_BINARY_DIR}/host/bin/clang++" "${CMAKE_BINARY_DIR}/host/bin/wasm-ld"
        "${CMAKE_BINARY_DIR}/host/bin/llvm-ar" "${CMAKE_BINARY_DIR}/host/bin/llvm-ranlib" "${CMAKE_BINARY_DIR}/host/bin/llvm-nm")
message(STATUS "Use -DRL_TOOLS_IDE_HOST_LLVM_BIN=${CMAKE_BINARY_DIR}/host/bin in the IDE toolchain configure.")
