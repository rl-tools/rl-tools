find_program(RL_TOOLS_NODE_EXECUTABLE node)
find_program(RL_TOOLS_PYTHON3_EXECUTABLE python3)
find_program(RL_TOOLS_BROWSER_EXECUTABLE NAMES google-chrome google-chrome-stable chromium chromium-browser firefox)
set(RL_TOOLS_IDE_TOOLCHAIN_PACKAGE "${CMAKE_SOURCE_DIR}/static/ide/build/toolchain")
set(RL_TOOLS_IDE_ARTIFACTS_PRESENT FALSE)
if(EXISTS "${RL_TOOLS_IDE_TOOLCHAIN_PACKAGE}/llvm.wasm" AND EXISTS "${RL_TOOLS_IDE_TOOLCHAIN_PACKAGE}/sysroot.tar" AND EXISTS "${CMAKE_SOURCE_DIR}/static/ide/build/rl_tools_include.tar")
    set(RL_TOOLS_IDE_ARTIFACTS_PRESENT TRUE)
endif()

if(RL_TOOLS_NODE_EXECUTABLE)
    add_test(NAME test_ide_unit COMMAND ${RL_TOOLS_NODE_EXECUTABLE} --test ${CMAKE_CURRENT_SOURCE_DIR}/wasi.test.mjs ${CMAKE_CURRENT_SOURCE_DIR}/tar.test.mjs ${CMAKE_CURRENT_SOURCE_DIR}/protocol.test.mjs)
    set_tests_properties(test_ide_unit PROPERTIES LABELS "ide" TIMEOUT 120)
else()
    message(STATUS "RLtools IDE unit tests disabled (node not found)")
endif()

if(RL_TOOLS_IDE_ARTIFACTS_PRESENT)
    add_test(NAME test_ide_toolchain_provenance COMMAND ${CMAKE_CURRENT_SOURCE_DIR}/provenance.sh ${RL_TOOLS_IDE_TOOLCHAIN_PACKAGE})
    set_tests_properties(test_ide_toolchain_provenance PROPERTIES LABELS "ide" TIMEOUT 60)
    if(RL_TOOLS_NODE_EXECUTABLE)
        add_test(NAME test_ide_toolchain COMMAND ${RL_TOOLS_NODE_EXECUTABLE} --no-warnings ${CMAKE_CURRENT_SOURCE_DIR}/toolchain.mjs --toolchain ${RL_TOOLS_IDE_TOOLCHAIN_PACKAGE})
        set_tests_properties(test_ide_toolchain PROPERTIES LABELS "ide" TIMEOUT 300)
        add_test(NAME test_ide_pipeline COMMAND ${RL_TOOLS_NODE_EXECUTABLE} --no-warnings ${CMAKE_CURRENT_SOURCE_DIR}/pipeline.mjs)
        set_tests_properties(test_ide_pipeline PROPERTIES LABELS "ide" TIMEOUT 600)
    endif()
    if(RL_TOOLS_PYTHON3_EXECUTABLE AND RL_TOOLS_BROWSER_EXECUTABLE)
        add_test(NAME test_ide_browser COMMAND ${RL_TOOLS_PYTHON3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/browser_test.py --browser ${RL_TOOLS_BROWSER_EXECUTABLE} --root ${CMAKE_SOURCE_DIR})
        set_tests_properties(test_ide_browser PROPERTIES LABELS "ide" TIMEOUT 900)
    else()
        message(STATUS "RLtools IDE browser test disabled (needs python3 and google-chrome/chromium/firefox)")
    endif()
else()
    message(STATUS "RLtools IDE toolchain/pipeline/browser tests disabled (needs static/ide/build from the tools/ide/toolchain superbuild + tools/ide/bundle.sh)")
endif()
